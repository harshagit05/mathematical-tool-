<script>
(function(){
  const CITIES = [
    ["San Francisco","America/Los_Angeles"],["Seattle","America/Los_Angeles"],
    ["Denver","America/Denver"],["Chicago","America/Chicago"],
    ["New York","America/New_York"],["Miami","America/New_York"],
    ["Toronto","America/Toronto"],["Mexico City","America/Mexico_City"],
    ["Bogotá","America/Bogota"],["Lima","America/Lima"],
    ["Santiago","America/Santiago"],["São Paulo","America/Sao_Paulo"],
    ["Buenos Aires","America/Argentina/Buenos_Aires"],["Reykjavik","Atlantic/Reykjavik"],
    ["London","Europe/London"],["Lisbon","Europe/Lisbon"],
    ["Paris","Europe/Paris"],["Madrid","Europe/Madrid"],
    ["Amsterdam","Europe/Amsterdam"],["Berlin","Europe/Berlin"],
    ["Zurich","Europe/Zurich"],["Rome","Europe/Rome"],
    ["Warsaw","Europe/Warsaw"],["Athens","Europe/Athens"],
    ["Helsinki","Europe/Helsinki"],["Moscow","Europe/Moscow"],
    ["Istanbul","Europe/Istanbul"],["Cairo","Africa/Cairo"],
    ["Lagos","Africa/Lagos"],["Johannesburg","Africa/Johannesburg"],
    ["Nairobi","Africa/Nairobi"],["Tel Aviv","Asia/Jerusalem"],
    ["Riyadh","Asia/Riyadh"],["Dubai","Asia/Dubai"],
    ["Karachi","Asia/Karachi"],["Mumbai","Asia/Kolkata"],
    ["Delhi","Asia/Kolkata"],["Colombo","Asia/Colombo"],
    ["Kathmandu","Asia/Kathmandu"],["Dhaka","Asia/Dhaka"],
    ["Bangkok","Asia/Bangkok"],["Jakarta","Asia/Jakarta"],
    ["Ho Chi Minh City","Asia/Ho_Chi_Minh"],["Singapore","Asia/Singapore"],
    ["Kuala Lumpur","Asia/Kuala_Lumpur"],["Manila","Asia/Manila"],
    ["Hong Kong","Asia/Hong_Kong"],["Shanghai","Asia/Shanghai"],
    ["Beijing","Asia/Shanghai"],["Seoul","Asia/Seoul"],
    ["Tokyo","Asia/Tokyo"],["Perth","Australia/Perth"],
    ["Adelaide","Australia/Adelaide"],["Sydney","Australia/Sydney"],
    ["Melbourne","Australia/Melbourne"],["Auckland","Pacific/Auckland"],
    ["Honolulu","Pacific/Honolulu"],["Anchorage","America/Anchorage"],
    ["Vancouver","America/Vancouver"]
  ];

  const dl = document.getElementById('cityList');
  CITIES.forEach(([name]) => {
    const opt = document.createElement('option');
    opt.value = name;
    dl.appendChild(opt);
  });

  const localTz = Intl.DateTimeFormat().resolvedOptions().timeZone;
  function friendlyFromTz(tz){
    const parts = tz.split('/');
    return (parts[parts.length-1] || tz).replace(/_/g,' ');
  }

  let zones = [];
  const seedNames = ["London","New York","Tokyo","Sydney"];
  const localAlready = CITIES.find(c => c[1] === localTz);
  zones.push({ label: localAlready ? localAlready[0] : friendlyFromTz(localTz), tz: localTz, isLocal: true });
  seedNames.forEach(n => {
    const found = CITIES.find(c => c[0] === n);
    if (found && found[1] !== localTz) zones.push({ label: found[0], tz: found[1], isLocal:false });
  });

  let baseTimestamp = Date.now();
  let liveMode = true;
  let tickHandle = null;

  function getOffsetMinutes(timeZone, date){
    const dtf = new Intl.DateTimeFormat('en-US', {
      timeZone, hour12:false,
      year:'numeric', month:'2-digit', day:'2-digit',
      hour:'2-digit', minute:'2-digit', second:'2-digit'
    });
    const parts = dtf.formatToParts(date).reduce((acc,p)=>{ if(p.type!=='literal') acc[p.type]=p.value; return acc; },{});
    let hour = parseInt(parts.hour,10);
    if (hour === 24) hour = 0;
    const asUTC = Date.UTC(
      parseInt(parts.year,10), parseInt(parts.month,10)-1, parseInt(parts.day,10),
      hour, parseInt(parts.minute,10), parseInt(parts.second,10)
    );
    return Math.round((asUTC - date.getTime())/60000);
  }

  function formatOffset(mins){
    const sign = mins >= 0 ? '+' : '-';
    const abs = Math.abs(mins);
    const h = Math.floor(abs/60);
    const m = abs % 60;
    return `UTC${sign}${h}${m ? ':' + String(m).padStart(2,'0') : ''}`;
  }

  function localParts(tz, date){
    const dtf = new Intl.DateTimeFormat('en-US', {
      timeZone: tz, hour12:false,
      year:'numeric', month:'2-digit', day:'2-digit',
      hour:'2-digit', minute:'2-digit', second:'2-digit', weekday:'short'
    });
    const parts = dtf.formatToParts(date).reduce((acc,p)=>{ if(p.type!=='literal') acc[p.type]=p.value; return acc; },{});
    let hour = parseInt(parts.hour,10);
    if (hour === 24) hour = 0;
    return {
      hour, minute: parseInt(parts.minute,10), second: parseInt(parts.second,10),
      day: parseInt(parts.day,10), month: parseInt(parts.month,10), year: parseInt(parts.year,10),
      weekday: parts.weekday
    };
  }

  function dayIndex(p){
    return Date.UTC(p.year, p.month-1, p.day) / 86400000;
  }

  function fmt12(hour, minute){
    const ampm = hour >= 12 ? 'PM' : 'AM';
    let h = hour % 12; if (h === 0) h = 12;
    return { h, m: String(minute).padStart(2,'0'), ampm };
  }

  const rowsEl = document.getElementById('rows');
  const heroTimeEl = document.getElementById('heroTime');
  const heroMetaEl = document.getElementById('heroMeta');
  const liveBtn = document.getElementById('liveBtn');
  const liveLabel = document.getElementById('liveLabel');

  const HOUR_LABELS = [0,3,6,9,12,15,18,21];

  function render(){
    const date = new Date(baseTimestamp);
    const refParts = localParts(zones[0].tz, date);
    const refDayIdx = dayIndex(refParts);

    // Hero
    const h12 = fmt12(refParts.hour, refParts.minute);
    heroTimeEl.innerHTML = `${String(h12.h).padStart(2,'0')}:${h12.m}<span class="sec">:${String(refParts.second).padStart(2,'0')} ${h12.ampm}</span>`;
    heroMetaEl.innerHTML = `<b>${zones[0].label}</b> · ${refParts.weekday} ${refParts.month}/${refParts.day}/${refParts.year} · ${formatOffset(getOffsetMinutes(zones[0].tz, date))}`;

    rowsEl.innerHTML = '';
    zones.forEach((z, idx) => {
      const p = localParts(z.tz, date);
      const frac = (p.hour + p.minute/60 + p.second/3600) / 24;
      const offMin = getOffsetMinutes(z.tz, date);
      const dDiff = dayIndex(p) - refDayIdx;
      const h12z = fmt12(p.hour, p.minute);

      const row = document.createElement('div');
      row.className = 'row';
      row.innerHTML = `
        <div class="row-info">
          <div class="row-city">${z.label}${idx===0 ? '<span class="ref-tag">REF</span>' : ''}</div>
          <div class="row-sub">${formatOffset(offMin)}${dDiff !== 0 ? `<span class="daydiff">${dDiff > 0 ? '+' : ''}${dDiff}d</span>` : ''}</div>
        </div>
        <div class="row-main">
          <div class="strip" data-idx="${idx}">
            <div class="ticks">${HOUR_LABELS.map(()=>'<div class="tick"></div>').join('')}</div>
            ${HOUR_LABELS.map(hh => `<div class="label" style="left:${(hh/24*100)}%">${hh===0?'12a':hh===12?'12p':hh<12?hh+'a':(hh-12)+'p'}</div>`).join('')}
            <div class="needle" style="left:${frac*100}%"></div>
          </div>
          <div class="row-digital">
            <span class="time">${String(h12z.h).padStart(2,'0')}:${h12z.m}</span><span class="ampm">${h12z.ampm}</span>
          </div>
        </div>
        ${idx===0 ? '' : `<button class="remove-btn" data-remove="${idx}" title="Remove">✕</button>`}
      `;
      rowsEl.appendChild(row);
    });

    rowsEl.querySelectorAll('.strip').forEach(attachDrag);
    rowsEl.querySelectorAll('[data-remove]').forEach(btn => {
      btn.addEventListener('click', () => {
        zones.splice(parseInt(btn.dataset.remove,10), 1);
        render();
      });
    });
  }

  function attachDrag(el){
    let dragging = false;
    let startX = 0;
    let startTs = 0;

    function down(clientX){
      dragging = true;
      startX = clientX;
      startTs = baseTimestamp;
      setLive(false);
    }
    function move(clientX){
      if (!dragging) return;
      const rect = el.getBoundingClientRect();
      const deltaX = clientX - startX;
      const deltaHours = (deltaX / rect.width) * 24;
      baseTimestamp = startTs + deltaHours * 3600000;
      render();
    }
    function up(){ dragging = false; }

    el.addEventListener('pointerdown', e => { el.setPointerCapture(e.pointerId); down(e.clientX); });
    el.addEventListener('pointermove', e => move(e.clientX));
    el.addEventListener('pointerup', up);
    el.addEventListener('pointercancel', up);
  }

  function setLive(on){
    liveMode = on;
    liveBtn.classList.toggle('is-live', on);
    liveLabel.textContent = on ? 'Live' : 'Now';
    if (on){
      baseTimestamp = Date.now();
      render();
    }
  }

  liveBtn.addEventListener('click', () => setLive(true));

  if (tickHandle) clearInterval(tickHandle);
  tickHandle = setInterval(() => {
    if (liveMode){
      baseTimestamp = Date.now();
      render();
    }
  }, 1000);

  const cityInput = document.getElementById('cityInput');
  const addBtn = document.getElementById('addBtn');

  function addCity(){
    const val = cityInput.value.trim();
    if (!val) return;
    const found = CITIES.find(c => c[0].toLowerCase() === val.toLowerCase());
    if (!found){
      cityInput.style.borderColor = 'var(--danger)';
      setTimeout(() => { cityInput.style.borderColor = ''; }, 900);
      return;
    }
    zones.push({ label: found[0], tz: found[1], isLocal:false });
    cityInput.value = '';
    render();
  }

  addBtn.addEventListener('click', addCity);
  cityInput.addEventListener('keydown', e => { if (e.key === 'Enter') addCity(); });

  render();
})();
</script>
